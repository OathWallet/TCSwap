Building production-ready applications with TCSwap requires careful attention to security, performance, and user experience. This guide covers essential best practices.

## Security

### API Key Management

For frontend applications, use a backend proxy to keep API keys secure:

```typescript
// @noErrorValidation

const quote = await fetch("/api/swap-quote", {
  method: "POST",
  body: JSON.stringify(quoteParams),
}).then((r) => r.json());

import { USwapConfig } from "@tcswap/helpers";
import { USwapApi } from "@tcswap/helpers/api";

// Server only: the key is sent as the `x-api-key` header
USwapConfig.setApiKey("uSwap", process.env.USWAP_API_KEY);

export async function POST(request: Request) {
  const params = await request.json();

  const quote = await USwapApi.getSwapQuote(params);

  return Response.json(quote);
}
```

### Private Key Handling

Always use secure wallet connections instead of handling private keys directly:

```typescript

import { createUSwap, Chain, NetworkDerivationPath } from "@tcswap/sdk";

const uSwap = createUSwap();

await uSwap.connectEVMWallet([Chain.Ethereum]);
await uSwap.connectLedger([Chain.Bitcoin]);
await uSwap.connectTrezor([Chain.Bitcoin], NetworkDerivationPath[Chain.Bitcoin]);
```

### Input Validation

Always validate user inputs:

```typescript
// @noErrorValidation
import { getAddressValidator, USwapError } from "@tcswap/sdk";

const validateAddress = await getAddressValidator();

function validateSwapInput(params: any): void {
  if (!validateAddress({ address: params.recipient, chain: params.chain })) {
    throw new USwapError({
      errorKey: "core_swap_invalid_params",
      info: { message: "Invalid recipient address" },
    });
  }

  const amount = Number(params.amount);
  if (isNaN(amount) || amount <= 0) {
    throw new USwapError({
      errorKey: "core_swap_invalid_params",
      info: { message: "Invalid amount" },
    });
  }

  if (params.slippage < 0 || params.slippage > 50) {
    throw new USwapError({
      errorKey: "core_swap_invalid_params",
      info: { message: "Slippage must be between 0 and 50%" },
    });
  }
}
```

### Lazy Loading Toolboxes

Load toolboxes only when needed:

```typescript
// @noErrorValidation
import { Chain } from "@tcswap/sdk";

async function getToolboxForChain(chain: Chain) {
  switch (chain) {
    case Chain.Ethereum: {
      const { getEvmToolbox } = await import("@tcswap/toolboxes/evm");
      return getEvmToolbox(chain);
    }

    case Chain.Bitcoin: {
      const { getUtxoToolbox } = await import("@tcswap/toolboxes/utxo");
      return getUtxoToolbox(chain);
    }

    default: {
      const { getToolbox } = await import("@tcswap/toolboxes");
      return getToolbox(chain);
    }
  }
}
```

### Request Caching

Implement caching for frequently accessed data:

```typescript
// @noErrorValidation
class USwapCache {
  private cache = new Map<string, { data: any; timestamp: number }>();
  private ttl: Record<string, number> = {
    price: 60_000,
    balance: 300_000,
    quote: 30_000,
  };

  async get<T>(
    key: string,
    type: keyof typeof this.ttl,
    fetcher: () => Promise<T>
  ): Promise<T> {
    const cached = this.cache.get(key);
    const now = Date.now();

    if (cached && now - cached.timestamp < this.ttl[type]) {
      return cached.data;
    }

    const data = await fetcher();
    this.cache.set(key, { data, timestamp: now });
    return data;
  }
}

const cache = new USwapCache();

const price = await cache.get(`price_${asset}`, "price", async () => {
  const { USwapApi } = await import("@tcswap/helpers/api");
  return USwapApi.getPrice({ tokens: [{ identifier: asset }], metadata: false });
});
```

## Error Handling

### Comprehensive Error Handling

Handle all error scenarios gracefully:

```typescript
// @noErrorValidation
import { USwapError, type QuoteResponseRoute } from "@tcswap/sdk";

async function executeSwap(route: QuoteResponseRoute) {
  try {
    const txHash = await uSwap.swap({ route });
    return { success: true, txHash };
  } catch (error) {
    if (error instanceof USwapError) {
      // error.message starts with the error key (followed by ": {info}" when present)
      const [errorKey] = error.message.split(":");

      switch (errorKey) {
        case "core_wallet_connection_not_found":
          return { success: false, error: "Please connect your wallet" };

        case "toolbox_utxo_insufficient_balance":
        case "core_transaction_deposit_insufficient_funds_error":
          return { success: false, error: "Insufficient balance" };

        case "core_transaction_user_rejected":
        case "wallet_connection_rejected_by_user":
          return { success: false, error: "Transaction cancelled" };

        default:
          return { success: false, error: error.message };
      }
    }

    if (error instanceof Error && error.message.includes("network")) {
      return { success: false, error: "Network error. Please try again." };
    }

    console.error("Unexpected error:", error);
    return { success: false, error: "An unexpected error occurred" };
  }
}
```

### Retry Logic

Implement retry logic for transient failures:

```typescript
// @noErrorValidation
import { USwapError, USwapApi } from "@tcswap/sdk";

async function withRetry<T>(
  fn: () => Promise<T>,
  options = { maxRetries: 3, delay: 1000 }
): Promise<T> {
  let lastError: Error;

  for (let i = 0; i <= options.maxRetries; i++) {
    try {
      return await fn();
    } catch (error) {
      lastError = error as Error;

      if (
        error instanceof USwapError &&
        error.message.startsWith("core_transaction_user_rejected")
      ) {
        throw error;
      }

      if (i < options.maxRetries) {
        await new Promise((r) => setTimeout(r, options.delay * (i + 1)));
      }
    }
  }

  throw lastError!;
}

const quote = await withRetry(() => USwapApi.getSwapQuote(params));
```

## User Experience

### Loading States

Provide clear feedback during operations:

```typescript
// @noErrorValidation
import { USwapApi, type AssetValue, type QuoteRequest } from "@tcswap/sdk";

interface SwapState {
  status: "idle" | "quoting" | "approving" | "swapping" | "success" | "error";
  message?: string;
  txHash?: string;
}

async function performSwap(params: QuoteRequest, assetValue: AssetValue) {
  const setState = (update: Partial<SwapState>) => {};

  try {
    setState({ status: "quoting", message: "Getting best price..." });
    const quote = await USwapApi.getSwapQuote(params);

    const route = quote.routes[0];

    // Spender: the contract the route's EVM tx calls, or the router (targetAddress)
    const spender =
      route.tx && typeof route.tx === "object" && "to" in route.tx
        ? route.tx.to
        : route.targetAddress;

    if (spender) {
      setState({ status: "approving", message: "Approving tokens..." });
      // Resolves immediately for native gas assets and non-EVM chains
      await uSwap.approveAssetValue(assetValue, spender);
    }

    setState({ status: "swapping", message: "Executing swap..." });
    const txHash = await uSwap.swap({ route });

    setState({ status: "success", txHash });
  } catch (error) {
    setState({
      status: "error",
      message: getErrorMessage(error),
    });
  }
}
```

### Transaction Tracking

Track transaction status for better UX:

```typescript
// @noErrorValidation
import { ChainId, USwapApi } from "@tcswap/sdk";

async function trackTransaction(
  hash: string,
  chainId: ChainId,
  onUpdate: (status: string) => void
) {
  let attempts = 0;
  const maxAttempts = 60;

  while (attempts < maxAttempts) {
    try {
      const details = await USwapApi.getTrackerDetails({ hash, chainId });

      onUpdate(details.status);

      if (["completed", "refunded", "failed"].includes(details.status)) {
        return details;
      }
    } catch (error) {}

    await new Promise((r) => setTimeout(r, 5000));
    attempts++;
  }

  throw new Error("Transaction tracking timeout");
}
```

## Network Configuration

### RPC Failover

Configure multiple RPC endpoints for reliability. The first URL is the primary; if it fails a connection test, TCSwap tries the remaining URLs in order:

```typescript
// @noErrorValidation
import { Chain, USwapConfig } from "@tcswap/sdk";

USwapConfig.setRpcUrl(
  Chain.Ethereum,
  [
    process.env.ALCHEMY_ETH_RPC,
    process.env.INFURA_ETH_RPC,
    "https://ethereum-rpc.publicnode.com",
  ].filter(Boolean) as string[]
);

USwapConfig.setRpcUrl(
  Chain.Avalanche,
  [process.env.AVALANCHE_RPC, "https://api.avax.network/ext/bc/C/rpc"].filter(
    Boolean
  ) as string[]
);
```

## Monitoring & Analytics

### Error Tracking

Track errors for debugging:

```typescript
// @noErrorValidation
import * as Sentry from "@sentry/browser";

Sentry.init({
  dsn: process.env.SENTRY_DSN,
  environment: process.env.NODE_ENV,

    if (event.request?.data) {
      delete event.request.data.privateKey;
      delete event.request.data.mnemonic;
    }
    return event;
  },

try {
  await uSwap.swap({ route });

  Sentry.captureException(error, {
    tags: {
      chain: route.sellAsset.split(".")[0],
      provider: route.providers[0],
```

### Performance Monitoring

Track key metrics:

```typescript
// @noErrorValidation
class PerformanceTracker {
  track(operation: string, fn: () => Promise<any>) {
    const start = performance.now();

    return fn()
      .then((result) => {
        const duration = performance.now() - start;
        this.report(operation, duration, "success");
        return result;
      })
      .catch((error) => {
        const duration = performance.now() - start;
        this.report(operation, duration, "error");
        throw error;
      });
  }

  private report(operation: string, duration: number, status: string) {
    analytics.track("operation_performance", {
      operation,
      duration,
      status,
    });
  }
}

const tracker = new PerformanceTracker();

const txHash = await tracker.track("swap", () => uSwap.swap({ route }));
```
