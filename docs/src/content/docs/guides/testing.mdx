## Unit Testing

### Testing TCSwap Configuration

Test configuration management:

```typescript
// @noErrorValidation

import { describe, expect, test, beforeEach } from 'bun:test';
import { USwapConfig } from '@tcswap/helpers';
import { Chain } from '@tcswap/types';

describe('USwap Configuration', () => {
  beforeEach(() => {
    USwapConfig.reinitialize();
  });

  test('should set and get API keys', () => {
    const apiKey = 'test-api-key-123';

    USwapConfig.setApiKey('uSwap', apiKey);

    const config = USwapConfig.get('apiKeys');
    expect(config.uSwap).toBe(apiKey);
  });

  test('should set RPC URLs', () => {
    const customUrls = ['https://ethereum-rpc.publicnode.com'];

    USwapConfig.setRpcUrl(Chain.Ethereum, customUrls);

    const rpcUrls = USwapConfig.get('rpcUrls');
    expect(rpcUrls[Chain.Ethereum]).toEqual(customUrls);
  });

  test('should preserve existing config when adding new keys', () => {
    USwapConfig.setApiKey('uSwap', 'key1');
    USwapConfig.setApiKey('walletConnectProjectId', 'key2');

    const apiKeys = USwapConfig.get('apiKeys');
    expect(apiKeys.uSwap).toBe('key1');
    expect(apiKeys.walletConnectProjectId).toBe('key2');
  });
});
```

### Testing Asset Value Operations

Test asset value calculations:

```typescript
// @noErrorValidation

import { describe, expect, test } from "bun:test";
import { AssetValue } from "@tcswap/helpers";

describe("AssetValue", () => {
  test("should create AssetValue from string", () => {
    const assetValue = AssetValue.from({ asset: "ETH.ETH", value: "1.5" });

    expect(assetValue.toString()).toBe("ETH.ETH");
    expect(assetValue.getValue("string")).toBe("1.5");
    expect(assetValue.chain).toBe("ETH");
  });

  test("should handle decimal precision", () => {
    const assetValue = AssetValue.from({ asset: "BTC.BTC", value: "0.12345678" });

    expect(assetValue.getValue("string", 8)).toBe("0.12345678");
    expect(assetValue.getValue("string", 4)).toBe("0.1235");
  });

  test("should perform arithmetic operations", () => {
    const asset1 = AssetValue.from({ asset: "ETH.ETH", value: "1.5" });
    const asset2 = AssetValue.from({ asset: "ETH.ETH", value: "0.5" });

    const sum = asset1.add(asset2);
    expect(sum.getValue("string")).toBe("2");

    const difference = asset1.sub(asset2);
    expect(difference.getValue("string")).toBe("1");
  });

  test("should handle different asset types", () => {
    const eth = AssetValue.from({ asset: "ETH.ETH", value: "1" });
    const usdc = AssetValue.from({
      asset: "ETH.USDC-0xA0b86991c6218b36c1d19D4a2e9Eb0cE3606eB48",
      value: "1000",
    });

    expect(eth.chain).toBe("ETH");
    expect(eth.isGasAsset).toBe(true);
    expect(usdc.chain).toBe("ETH");
    expect(usdc.ticker).toBe("USDC");
  });

  test("should reject invalid asset URLs", () => {
    expect(() => AssetValue.fromUrl("")).toThrow("helpers_invalid_asset_url");
  });
});
```

### Testing TCSwap Error Handling

Test error creation and handling:

```typescript
// @noErrorValidation

import { describe, expect, test } from "bun:test";
import { USwapError } from "@tcswap/helpers";

describe("USwapError", () => {
  test("should create error with string key", () => {
    const error = new USwapError("core_wallet_connection_not_found");

    expect(error).toBeInstanceOf(USwapError);
    expect(error.name).toBe("USwapError");
    expect(error.message).toBe("core_wallet_connection_not_found");
  });

  test("should create error with additional info", () => {
    const errorInfo = { chainId: 1, address: "0x123..." };
    const error = new USwapError({
      errorKey: "wallet_evm_extensions_failed_to_switch_network",
      info: errorInfo,
    });

    expect(error.message).toContain(
      "wallet_evm_extensions_failed_to_switch_network"
    );
    expect(error.message).toContain(JSON.stringify(errorInfo));
  });

  test("should preserve cause", () => {
    const originalError = new Error("Network timeout");
    const uSwapError = new USwapError("api_v2_server_error", originalError);

    expect(uSwapError.cause).toBe(originalError);
  });

  test("should have correct error codes", () => {
    expect(USwapError.ErrorCode.core_wallet_connection_not_found).toBe(10101);
    expect(USwapError.ErrorCode.wallet_ledger_device_not_found).toBe(20104);
    expect(USwapError.ErrorCode.api_v2_server_error).toBe(60002);
  });
});
```

### Testing Utility Functions

Test helper functions:

```typescript
// @noErrorValidation

import { describe, expect, test } from "bun:test";
import { validateIdentifier } from "@tcswap/helpers";
import { getAddressValidator } from "@tcswap/toolboxes";
import { Chain } from "@tcswap/types";

describe("Validators", async () => {
  const validateAddress = await getAddressValidator();

  test("should validate Ethereum addresses", () => {
    const validAddress = "0x742d35Cc6486C0D8C50fb5F04F08E8b23e20Ab8A";
    const invalidAddress = "0xinvalid";

    expect(validateAddress({ address: validAddress, chain: Chain.Ethereum })).toBe(true);
    expect(validateAddress({ address: invalidAddress, chain: Chain.Ethereum })).toBe(false);
  });

  test("should validate Bitcoin addresses", () => {
    const validBech32 = "bc1qxy2kgdygjrsqtzq2n0yrf2493p83kkfjhx0wlh";
    const validLegacy = "1A1zP1eP5QGefi2DMPTfTL5SLmv7DivfNa";
    const invalid = "invalid-bitcoin-address";

    expect(validateAddress({ address: validBech32, chain: Chain.Bitcoin })).toBe(true);
    expect(validateAddress({ address: validLegacy, chain: Chain.Bitcoin })).toBe(true);
    expect(validateAddress({ address: invalid, chain: Chain.Bitcoin })).toBe(false);
  });

  test("should validate asset identifiers", () => {
    expect(validateIdentifier("ETH.ETH")).toBe(true);
    expect(validateIdentifier("BTC.BTC")).toBe(true);
    expect(
      validateIdentifier("ETH.USDC-0xA0b86991c6218b36c1d19D4a2e9Eb0cE3606eB48")
    ).toBe(true);
    // Throws USwapError("helpers_invalid_identifier") for unknown chains
    expect(() => validateIdentifier("invalid")).toThrow("helpers_invalid_identifier");
  });

  test("should validate chain-specific addresses", () => {
    const addresses = {
      [Chain.Ethereum]: "0x742d35Cc6486C0D8C50fb5F04F08E8b23e20Ab8A",
      [Chain.Bitcoin]: "bc1qxy2kgdygjrsqtzq2n0yrf2493p83kkfjhx0wlh",
      [Chain.THORChain]: "thor1x96cfrxhe2qfvs5qqe3t9z8gk8chkwaqmhc4vj",
      [Chain.Cosmos]: "cosmos1x96cfrxhe2qfvs5qqe3t9z8gk8chkwaqmhc4vj",
    };

    Object.entries(addresses).forEach(([chain, address]) => {
      expect(validateAddress({ address, chain: chain as Chain })).toBe(true);
    });
  });
});
```

## Integration Testing

### Testing Wallet Connections

Test wallet connection functionality:

```typescript
// @noErrorValidation

import { describe, expect, test, beforeEach } from "bun:test";
import { USwap } from "@tcswap/core";
import { keystoreWallet } from "@tcswap/wallets/keystore";
import { Chain } from "@tcswap/types";

const createClient = () => USwap({ wallets: { ...keystoreWallet } });

describe("Wallet Integration", () => {
  let uSwap: ReturnType<typeof createClient>;

  beforeEach(() => {
    uSwap = createClient();
  });

  test(
    "should connect keystore wallet",
    async () => {
      if (!process.env.TEST_PHRASE) {
        console.warn("TEST_PHRASE not set, skipping keystore test");
        return;
      }

      const chains = [Chain.Ethereum, Chain.Bitcoin];
      await uSwap.connectKeystore(chains, process.env.TEST_PHRASE);

      const wallets = uSwap.getAllWallets();

      expect(wallets[Chain.Ethereum]).toBeDefined();
      expect(wallets[Chain.Ethereum].address).toBeDefined();
      expect(wallets[Chain.Bitcoin]).toBeDefined();
      expect(wallets[Chain.Bitcoin].address).toBeDefined();
    },
    { timeout: 30000 }
  );

  test(
    "should get wallet balance",
    async () => {
      if (!process.env.TEST_PHRASE) {
        console.warn("TEST_PHRASE not set, skipping balance test");
        return;
      }

      await uSwap.connectKeystore([Chain.Ethereum], process.env.TEST_PHRASE);

      const wallet = await uSwap.getWalletWithBalance(Chain.Ethereum);

      expect(wallet).toBeDefined();
      expect(wallet.balance).toBeDefined();
      expect(Array.isArray(wallet.balance)).toBe(true);
    },
    { timeout: 30000 }
  );

  test("should handle connection errors gracefully", async () => {
    const invalidPhrase = "invalid mnemonic phrase";

    await expect(
      uSwap.connectKeystore([Chain.Ethereum], invalidPhrase)
    ).rejects.toThrow();
  });
});
```

### Testing API Integration

Test API calls and responses:

```typescript
// @noErrorValidation

import { describe, expect, test } from "bun:test";
import { USwapApi } from "@tcswap/helpers/api";

describe("API Integration", () => {
  test(
    "should get asset price",
    async () => {
      const [price] = await USwapApi.getPrice({
        tokens: [{ identifier: "ETH.ETH" }],
        metadata: false,
      });

      expect(price).toBeDefined();
      expect(typeof price.price_usd).toBe("number");
      expect(price.price_usd).toBeGreaterThan(0);
    },
    { timeout: 10000 }
  );

  test(
    "should get swap quote",
    async () => {
      const quoteParams = {
        sellAsset: "ETH.ETH",
        buyAsset: "ETH.USDC-0xA0b86991c6218b36c1d19D4a2e9Eb0cE3606eB48",
        sellAmount: "1",
        sourceAddress: "0x742d35Cc6486C0D8C50fb5F04F08E8b23e20Ab8A",
        destinationAddress: "0x742d35Cc6486C0D8C50fb5F04F08E8b23e20Ab8A",
      };

      const quote = await USwapApi.getSwapQuote(quoteParams);

      expect(quote).toBeDefined();
      expect(quote.routes).toBeDefined();
      expect(Array.isArray(quote.routes)).toBe(true);
      expect(quote.routes.length).toBeGreaterThan(0);
    },
    { timeout: 15000 }
  );

  test("should reject invalid quote params", async () => {
    await expect(
      USwapApi.getSwapQuote({
        sellAsset: "INVALID.ASSET",
        buyAsset: "BTC.BTC",
        sellAmount: "1",
      })
    ).rejects.toThrow();
  });
});
```

### Testing Swap Operations

Test end-to-end swap functionality:

```typescript
// @noErrorValidation

import { describe, expect, test, beforeEach } from "bun:test";
import { USwapApi } from "@tcswap/helpers/api";
import { ProviderName } from "@tcswap/helpers";

describe("Swap Integration", () => {
  test(
    "should get swap quote for THORChain swap",
    async () => {
      const quote = await USwapApi.getSwapQuote({
        sellAsset: "ETH.ETH",
        buyAsset: "BTC.BTC",
        sellAmount: "0.1",
        sourceAddress: "0x742d35Cc6486C0D8C50fb5F04F08E8b23e20Ab8A",
        destinationAddress: "bc1qxy2kgdygjrsqtzq2n0yrf2493p83kkfjhx0wlh",
        providers: [ProviderName.THORCHAIN],
      });

      expect(quote).toBeDefined();
      expect(quote.routes).toBeDefined();
      expect(quote.routes.length).toBeGreaterThan(0);

      const route = quote.routes[0];
      expect(route.sellAsset).toBe("ETH.ETH");
      expect(route.buyAsset).toBe("BTC.BTC");
      expect(route.expectedBuyAmount).toBeDefined();
    },
    { timeout: 30000 }
  );

  test("should validate swap parameters", async () => {
    await expect(
      USwapApi.getSwapQuote({
        sellAsset: "INVALID.ASSET",
        buyAsset: "BTC.BTC",
        sellAmount: "1",
        destinationAddress: "invalid-address",
      })
    ).rejects.toThrow();
  });

  test(
    "should handle insufficient liquidity",
    async () => {
      // The API either throws (api_v2_server_error) or returns no routes
      const result = await USwapApi.getSwapQuote({
        sellAsset: "ETH.ETH",
        buyAsset: "BTC.BTC",
        sellAmount: "10000000",
        destinationAddress: "bc1qxy2kgdygjrsqtzq2n0yrf2493p83kkfjhx0wlh",
      }).catch(() => null);

      expect(result === null || result.routes.length === 0).toBe(true);
    },
    { timeout: 30000 }
  );
});
```

```typescript
  promise: Promise<T>,
  ms: number
): Promise<T> {
```

## Performance Testing

### Load Testing

Test performance under load:

```typescript
// @noErrorValidation

import { describe, expect, test } from "bun:test";
import { USwapApi } from "@tcswap/helpers/api";

describe("Performance Tests", () => {
  test(
    "should handle concurrent price requests",
    async () => {
      const assets = ["ETH.ETH", "BTC.BTC", "AVAX.AVAX", "GAIA.ATOM", "DOGE.DOGE"];
      const startTime = performance.now();

      const prices = await Promise.all(
        assets.map((identifier) =>
          USwapApi.getPrice({ tokens: [{ identifier }], metadata: false })
        )
      );

      const duration = performance.now() - startTime;

      expect(prices).toHaveLength(5);
      expect(prices.every(([p]) => typeof p?.price_usd === "number")).toBe(true);
      expect(duration).toBeLessThan(5000);
    },
    { timeout: 10000 }
  );

  test(
    "should handle rate limiting gracefully",
    async () => {
      const requests = Array(50)
        .fill(0)
        .map(() =>
          USwapApi.getPrice({ tokens: [{ identifier: "ETH.ETH" }], metadata: false })
        );

      const results = await Promise.allSettled(requests);
      const successful = results.filter((r) => r.status === "fulfilled").length;
      const failed = results.filter((r) => r.status === "rejected").length;

      expect(successful).toBeGreaterThan(0);

      console.log(`Successful: ${successful}, Failed: ${failed}`);
    },
    { timeout: 30000 }
  );
});
```

## Continuous Integration

### GitHub Actions Configuration

Set up automated testing:

```yaml
# .github/workflows/test.yml
name: Tests

on:
  push:
    branches: [main, develop]
  pull_request:
    branches: [main, develop]

jobs:
  test:
    runs-on: ubuntu-latest

    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0
          ref: ${{ github.head_ref }}

      - uses: oven-sh/setup-bun@v1
        with:
          bun-version: latest

      - name: Install dependencies
        run: bun install

      - name: Run unit tests
        run: bun test src/
        env:
          TEST_API_KEY: ${{ secrets.TEST_API_KEY }}

      - name: Run integration tests
        run: bun test integration/
        env:
          TEST_API_KEY: ${{ secrets.TEST_API_KEY }}
          TEST_PHRASE: ${{ secrets.TEST_PHRASE }}
        if: github.event_name == 'push'

      - name: Generate coverage
        run: bun test --coverage

      - name: Upload coverage
        uses: codecov/codecov-action@v3
```

## Best Practices

### Test Organization

- **Separate unit and integration tests**: Keep fast unit tests separate from slower integration tests
- **Use descriptive test names**: Clear test names make failures easier to understand
- **Group related tests**: Use `describe` blocks to organize tests logically
- **Test error conditions**: Always test both success and failure scenarios

### Mocking Guidelines

- **Mock external dependencies**: Don't make real API calls in unit tests
- **Keep mocks simple**: Complex mocks can hide bugs
- **Reset mocks between tests**: Prevent test interdependence
- **Mock at the right level**: Mock at component boundaries, not internal functions

### Test Data Management

- **Use realistic test data**: Test data should resemble production data
- **Avoid hardcoded values**: Use constants and generators for test data
- **Clean up after tests**: Reset state and clean up resources
- **Use environment variables**: Keep sensitive test data in environment variables

### Performance Considerations

- **Run unit tests frequently**: Fast tests enable rapid iteration
- **Limit integration test scope**: Focus on critical paths
- **Use timeouts appropriately**: Set realistic timeouts for async operations
- **Monitor test duration**: Keep test suites fast to maintain developer productivity
